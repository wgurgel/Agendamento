// ============================================
// SISTEMA DE AGENDAMENTO DE SALA DE VÍDEO
// Com Firebase Firestore
// ============================================

// Variáveis globais
let bookings = [];

// ============================================
// FUNÇÕES DE INICIALIZAÇÃO
// ============================================

/**
 * Inicializa o sistema quando a página carrega
 */
document.addEventListener('DOMContentLoaded', () => {
  loadBookings();
  setDefaultDate();
  setupEventListeners();
});

/**
 * Configura os eventos dos botões e formulários
 */
function setupEventListeners() {
  // Formulário de agendamento
  const bookingForm = document.getElementById('bookingForm');
  if (bookingForm) {
    bookingForm.addEventListener('submit', handleBookingSubmit);
  }

  // Botão de limpar tudo
  const clearBtn = document.getElementById('clearAllBtn');
  if (clearBtn) {
    clearBtn.addEventListener('click', clearAllBookings);
  }

  // Filtro de data
  const filterDate = document.getElementById('filterDate');
  if (filterDate) {
    filterDate.addEventListener('change', renderBookings);
  }
}

/**
 * Define a data padrão para hoje
 */
function setDefaultDate() {
  const today = new Date().toISOString().split('T')[0];
  const bookingDate = document.getElementById('bookingDate');
  const filterDate = document.getElementById('filterDate');
  
  if (bookingDate) bookingDate.value = today;
  if (filterDate) filterDate.value = today;
}

// ============================================
// FUNÇÕES DO FIREBASE
// ============================================

/**
 * Carrega os agendamentos do Firebase Firestore em tempo real
 */
function loadBookings() {
  db.collection(COLLECTION_NAME)
    .orderBy('date')
    .orderBy('startTime')
    .onSnapshot((snapshot) => {
      bookings = [];
      snapshot.forEach((doc) => {
        bookings.push({
          id: doc.id,
          ...doc.data()
        });
      });
      renderBookings();
      console.log('✅ Agendamentos carregados:', bookings.length);
    }, (error) => {
      console.error('❌ Erro ao carregar:', error);
      showModal('❌', 'Erro ao carregar agendamentos. Verifique o console.');
    });
}

/**
 * Adiciona um novo agendamento no Firebase
 */
async function addBooking(personName, date, startTime, endTime) {
  const booking = {
    personName: personName.trim(),
    date: date,
    startTime: startTime,
    endTime: endTime,
    status: 'Ocupado',
    createdAt: new Date().toISOString()
  };

  try {
    await db.collection(COLLECTION_NAME).add(booking);
    showModal('✅', 'Agendamento realizado com sucesso!');
  } catch (error) {
    console.error('❌ Erro ao salvar:', error);
    showModal('❌', 'Erro ao salvar agendamento.');
  }
}

/**
 * Cancela um agendamento no Firebase
 */
async function cancelBooking(id) {
  if (confirm('Tem certeza que deseja cancelar este agendamento?')) {
    try {
      await db.collection(COLLECTION_NAME).doc(id).delete();
      showModal('🗑️', 'Agendamento cancelado com sucesso!');
    } catch (error) {
      console.error('❌ Erro ao cancelar:', error);
      showModal('❌', 'Erro ao cancelar agendamento.');
    }
  }
}

/**
 * Limpa todos os agendamentos do Firebase
 */
async function clearAllBookings() {
  if (confirm('️ ATENÇÃO: Apagar TODOS os agendamentos?')) {
    if (confirm('Última chance. Deseja realmente apagar tudo?')) {
      try {
        const snapshot = await db.collection(COLLECTION_NAME).get();
        const batch = db.batch();
        snapshot.forEach((doc) => {
          batch.delete(doc.ref);
        });
        await batch.commit();
        showModal('️', 'Todos os agendamentos foram removidos!');
      } catch (error) {
        console.error('❌ Erro ao limpar:', error);
        showModal('❌', 'Erro ao limpar agendamentos.');
      }
    }
  }
}

// ============================================
// FUNÇÕES DE MANIPULAÇÃO DO FORMULÁRIO
// ============================================

/**
 * Manipula o envio do formulário de agendamento
 */
function handleBookingSubmit(event) {
  event.preventDefault();

  const personName = document.getElementById('personName').value;
  const date = document.getElementById('bookingDate').value;
  const startTime = document.getElementById('startTime').value;
  const endTime = document.getElementById('endTime').value;

  // Validações
  if (!personName || !date || !startTime || !endTime) {
    showModal('⚠️', 'Por favor, preencha todos os campos!');
    return;
  }

  if (startTime >= endTime) {
    showModal('⚠️', 'O horário de término deve ser maior que o horário de início!');
    return;
  }

  // Verifica se já existe agendamento conflitante
  if (hasTimeConflict(date, startTime, endTime)) {
    showModal('⚠️', 'Já existe um agendamento neste horário!');
    return;
  }

  // Adiciona o agendamento
  addBooking(personName, date, startTime, endTime);

  // Limpa o formulário
  document.getElementById('bookingForm').reset();
  setDefaultDate();
}

/**
 * Verifica se existe conflito de horário
 */
function hasTimeConflict(date, startTime, endTime) {
  return bookings.some(booking => {
    if (booking.date !== date || booking.status !== 'Ocupado') {
      return false;
    }

    return (
      (startTime >= booking.startTime && startTime < booking.endTime) ||
      (endTime > booking.startTime && endTime <= booking.endTime) ||
      (startTime <= booking.startTime && endTime >= booking.endTime)
    );
  });
}

// ============================================
// FUNÇÕES DE RENDERIZAÇÃO
// ============================================

/**
 * Renderiza a tabela de agendamentos
 */
function renderBookings() {
  const tableBody = document.querySelector('#bookingsTable tbody');
  if (!tableBody) return;

  tableBody.innerHTML = '';

  const filterDate = document.getElementById('filterDate');
  const selectedDate = filterDate ? filterDate.value : '';

  // Filtra os agendamentos pela data selecionada
  const filteredBookings = selectedDate
    ? bookings.filter(b => b.date === selectedDate)
    : bookings;

  if (filteredBookings.length === 0) {
    const row = document.createElement('tr');
    row.innerHTML = `
      <td colspan="5" style="text-align: center; padding: 2rem; color: #666;">
        Nenhum agendamento encontrado${selectedDate ? ' para esta data' : ''}.
      </td>
    `;
    tableBody.appendChild(row);
    return;
  }

  // Ordena por horário
  filteredBookings.sort((a, b) => a.startTime.localeCompare(b.startTime));

  // Renderiza cada agendamento
  filteredBookings.forEach(booking => {
    const row = createBookingRow(booking);
    tableBody.appendChild(row);
  });
}

/**
 * Cria uma linha da tabela para um agendamento
 */
function createBookingRow(booking) {
  const row = document.createElement('tr');

  // Formata a data para exibição
  const dateObj = new Date(booking.date + 'T00:00:00');
  const dateFormatted = dateObj.toLocaleDateString('pt-BR', {
    day: '2-digit',
    month: '2-digit',
    year: 'numeric'
  });

  row.innerHTML = `
    <td>${dateFormatted}</td>
    <td>${formatTime(booking.startTime)} - ${formatTime(booking.endTime)}</td>
    <td>${escapeHtml(booking.personName)}</td>
    <td><span class="status-badge status-${booking.status.toLowerCase()}">${booking.status}</span></td>
    <td>
      <button onclick="cancelBooking('${booking.id}')" 
              class="btn-cancel" 
              title="Cancelar agendamento">
        🗑️ Cancelar
      </button>
    </td>
  `;

  return row;
}

/**
 * Formata o horário para exibição (HH:MM)
 */
function formatTime(time) {
  return time.substring(0, 5);
}

/**
 * Escapa caracteres HTML para prevenir XSS
 */
function escapeHtml(text) {
  const div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;
}

/**
 * Exibe um modal de mensagem
 */
function showModal(icon, message) {
  // Cria o modal se não existir
  let modal = document.getElementById('messageModal');
  if (!modal) {
    modal = document.createElement('div');
    modal.id = 'messageModal';
    modal.className = 'modal';
    modal.innerHTML = `
      <div class="modal-content">
        <div class="modal-icon"></div>
        <div class="modal-message"></div>
        <button class="modal-close">OK</button>
      </div>
    `;
    document.body.appendChild(modal);

    // Fecha o modal ao clicar no botão OK ou fora dele
    modal.querySelector('.modal-close').addEventListener('click', () => {
      modal.style.display = 'none';
    });

    modal.addEventListener('click', (e) => {
      if (e.target === modal) {
        modal.style.display = 'none';
      }
    });
  }

  // Atualiza o conteúdo
  modal.querySelector('.modal-icon').textContent = icon;
  modal.querySelector('.modal-message').textContent = message;
  modal.style.display = 'flex';

  // Fecha automaticamente após 3 segundos
  setTimeout(() => {
    modal.style.display = 'none';
  }, 3000);
}
